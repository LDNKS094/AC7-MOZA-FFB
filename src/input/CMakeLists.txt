cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(AC7NativeInput LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use the x64 MSVC toolchain on Windows.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MINHOOK "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/minhook")
add_library(minhook STATIC "${MINHOOK}/src/buffer.c" "${MINHOOK}/src/hook.c"
    "${MINHOOK}/src/trampoline.c" "${MINHOOK}/src/hde/hde64.c")
target_include_directories(minhook PUBLIC "${MINHOOK}/include")
add_library(AC7NativeInput SHARED input.cpp)
target_compile_definitions(AC7NativeInput PRIVATE UNICODE _UNICODE DIRECTINPUT_VERSION=0x0800)
target_compile_options(AC7NativeInput PRIVATE /W4 /O2)
target_link_libraries(AC7NativeInput PRIVATE minhook dxguid)
set_target_properties(AC7NativeInput PROPERTIES PREFIX "" SUFFIX ".asi")
